// ============================================================================
// DATA MODELS
// ============================================================================

enum ProblemCategory {
  flatTyre,
  engineTrouble,
  batteryDead,
  chainBroken,
  brakeIssue,
  outOfFuel,
  electrical,
  generalCheck,
}

class ProblemItem {
  final ProblemCategory category;
  final String title;
  final String description;
  final IconData icon;
  final double estimatedCost;
  final int estimatedTimeMinutes;
  final String? badge;

  const ProblemItem({
    required this.category,
    required this.title,
    required this.description,
    required this.icon,
    required this.estimatedCost,
    required this.estimatedTimeMinutes,
    this.badge,
  });
}

class MechanicProfile {
  final String name;
  final String phone;
  final double rating;
  final int totalRescues;
  final String vehicleModel;
  final String vehiclePlate;
  final String avatarUrl;

  const MechanicProfile({
    required this.name,
    required this.phone,
    required this.rating,
    required this.totalRescues,
    required this.vehicleModel,
    required this.vehiclePlate,
    required this.avatarUrl,
  });
}

enum TrackingStatus {
  dispatched,
  onTheWay,
  arrived,
  inRepair,
  completed,
}

// ============================================================================
// MAIN NAVIGATION WRAPPER (State controller for Booking <-> Tracking)
// ============================================================================

class MainNavigationScreen extends StatefulWidget {
  const MainNavigationScreen({super.key});

  @override
  State<MainNavigationScreen> createState() => _MainNavigationScreenState();
}

class _MainNavigationScreenState extends State<MainNavigationScreen> {
  bool _isTracking = false;
  ProblemItem? _bookedProblem;
  String _selectedBike = 'Royal Enfield Classic 350';
  bool _isEmergency = false;

  final MechanicProfile _assignedMechanic = const MechanicProfile(
    name: 'Rajesh Sharma',
    phone: '+91 98765 43210',
    rating: 4.9,
    totalRescues: 384,
    vehicleModel: 'Hero Splendor (Toolbox Van)',
    vehiclePlate: 'DL 04 AB 8821',
    avatarUrl: 'https://images.unsplash.com/photo-1544005313-94ddf0286df2?w=150',
  );

  void _onBookMechanic(ProblemItem problem, String bike, bool emergency) {
    setState(() {
      _bookedProblem = problem;
      _selectedBike = bike;
      _isEmergency = emergency;
      _isTracking = true;
    });
  }

  void _onCancelBooking() {
    setState(() {
      _isTracking = false;
      _bookedProblem = null;
    });
  }

  @override
  Widget build(BuildContext context) {
    if (_isTracking && _bookedProblem != null) {
      return MechanicTrackingScreen(
        problem: _bookedProblem!,
        bikeModel: _selectedBike,
        isEmergency: _isEmergency,
        mechanic: _assignedMechanic,
        onCancel: _onCancelBooking,
      );
    }

    return CustomerBookingScreen(
      onBook: _onBookMechanic,
    );
  }
}

class MechanicTrackingScreen extends StatefulWidget {
  final ProblemItem problem;
  final String bikeModel;
  final bool isEmergency;
  final MechanicProfile mechanic;
  final VoidCallback onCancel;

  const MechanicTrackingScreen({
    super.key,
    required this.problem,
    required this.bikeModel,
    required this.isEmergency,
    required this.mechanic,
    required this.onCancel,
  });

  @override
  State<MechanicTrackingScreen> createState() => _MechanicTrackingScreenState();
}

class _MechanicTrackingScreenState extends State<MechanicTrackingScreen>
    with SingleTickerProviderStateMixin {
  late AnimationController _animController;
  TrackingStatus _status = TrackingStatus.onTheWay;
  double _distanceRemainingKm = 1.6;
  int _etaMinutes = 6;

  @override
  void initState() {
    super.initState();
    _animController = AnimationController(
      vsync: this,
      duration: const Duration(seconds: 14),
    )..repeat();

    // Listen to simulate gradual mechanic approach
    _animController.addListener(() {
      final progress = _animController.value;
      if (mounted) {
        setState(() {
          _distanceRemainingKm = (1.6 * (1.0 - progress)).clamp(0.1, 1.6);
          _etaMinutes = (6 * (1.0 - progress)).ceil().clamp(1, 6);
        });
      }
    });
  }

  @override
  void dispose() {
    _animController.dispose();
    super.dispose();
  }

  void _advanceSimulationStep() {
    setState(() {
      switch (_status) {
        case TrackingStatus.dispatched:
          _status = TrackingStatus.onTheWay;
          break;
        case TrackingStatus.onTheWay:
          _status = TrackingStatus.arrived;
          _distanceRemainingKm = 0.0;
          _etaMinutes = 0;
          break;
        case TrackingStatus.arrived:
          _status = TrackingStatus.inRepair;
          break;
        case TrackingStatus.inRepair:
          _status = TrackingStatus.completed;
          break;
        case TrackingStatus.completed:
          _status = TrackingStatus.onTheWay;
          break;
      }
    });
  }

  void _showCallDialog() {
    showDialog(
      context: context,
      builder: (ctx) => AlertDialog(
        title: Row(
          children: [
            const CircleAvatar(
              backgroundColor: Color(0xFFE65100),
              child: Icon(Icons.phone, color: Colors.white, size: 20),
            ),
            const SizedBox(width: 12),
            Text('Call ${widget.mechanic.name}'),
          ],
        ),
        content: Text('Simulated phone call to ${widget.mechanic.phone}. He is currently riding towards your location with the toolkit.'),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(ctx),
            child: const Text('End Call'),
          ),
          ElevatedButton(
            style: ElevatedButton.styleFrom(backgroundColor: Colors.green),
            onPressed: () {
              Navigator.pop(ctx);
              ScaffoldMessenger.of(context).showSnackBar(

// ============================================================================
// CUSTOM PAINTER: ROAD NETWORK, ANIMATED ROUTE & MOTORCYCLE MARKER
// ============================================================================

class RoadMapPainter extends CustomPainter {
  final double progress; // 0.0 to 1.0
  final TrackingStatus status;

  RoadMapPainter({required this.progress, required this.status});

  @override
  void paint(Canvas canvas, Size size) {
    // 1. Canvas Background (Stylized dark map terrain)
    final bgPaint = Paint()..color = const Color(0xFF0F172A);
    canvas.drawRect(Rect.fromLTWH(0, 0, size.width, size.height), bgPaint);

    // 2. Secondary roads grid
    final minorRoadPaint = Paint()
      ..color = const Color(0xFF1E293B)
      ..strokeWidth = 4.0
      ..style = PaintingStyle.stroke;

    // Horizontal & vertical grid roads
    for (double y = 80; y < size.height; y += 100) {
      canvas.drawLine(Offset(0, y), Offset(size.width, y), minorRoadPaint);
    }
    for (double x = 60; x < size.width; x += 110) {
      canvas.drawLine(Offset(x, 0), Offset(x, size.height), minorRoadPaint);
    }

    // 3. Main Highway / Avenue
    final highwayPaint = Paint()
      ..color = const Color(0xFF334155)
      ..strokeWidth = 14.0
      ..strokeCap = StrokeCap.round
      ..style = PaintingStyle.stroke;

    final highwayCenterPaint = Paint()
      ..color = const Color(0xFF475569)
      ..strokeWidth = 2.0
      ..style = PaintingStyle.stroke;

    final routePath = Path();
    // Route from Mechanic Starting Point to Customer Breakdown Point
    final startPoint = Offset(size.width * 0.15, size.height * 0.22);
    final bendPoint1 = Offset(size.width * 0.40, size.height * 0.22);
    final bendPoint2 = Offset(size.width * 0.40, size.height * 0.45);
    final endPoint = Offset(size.width * 0.82, size.height * 0.45);

    routePath.moveTo(startPoint.dx, startPoint.dy);
    routePath.lineTo(bendPoint1.dx, bendPoint1.dy);
    routePath.lineTo(bendPoint2.dx, bendPoint2.dy);
    routePath.lineTo(endPoint.dx, endPoint.dy);

    // Draw main avenue base
    canvas.drawPath(routePath, highwayPaint);
    canvas.drawPath(routePath, highwayCenterPaint);

    // 4. Highlighted Active Navigation Path (Emergency Orange Polyline)
    final activeRoutePaint = Paint()
      ..color = const Color(0xFFFF9800)
      ..strokeWidth = 6.0
      ..strokeCap = StrokeCap.round
      ..strokeJoin = StrokeJoin.round
      ..style = PaintingStyle.stroke;

    canvas.drawPath(routePath, activeRoutePaint);

    // 5. Customer Breakdown Location Pin & Radar Pulse
    final radarPaint = Paint()
      ..color = Colors.redAccent.withOpacity(0.25)
      ..style = PaintingStyle.fill;
    canvas.drawCircle(endPoint, 24, radarPaint);

    final customerPinBase = Paint()
      ..color = Colors.red
      ..style = PaintingStyle.fill;
    canvas.drawCircle(endPoint, 10, customerPinBase);

    final customerPinCenter = Paint()
      ..color = Colors.white
      ..style = PaintingStyle.fill;
    canvas.drawCircle(endPoint, 4, customerPinCenter);

    // Customer label badge
    final textPainter = TextPainter(
      text: const TextSpan(
        text: '📍 You (Stranded)',
        style: TextStyle(
          color: Colors.white,
          fontSize: 11,
          fontWeight: FontWeight.bold,
          backgroundColor: Color(0xCC000000),
        ),
      ),
      textDirection: TextDirection.ltr,
    );
    textPainter.layout();
    textPainter.paint(canvas, Offset(endPoint.dx - 45, endPoint.dy - 32));

    // 6. Calculate Moving Mechanic Position along the route
    // Total distance parts
    final seg1 = (bendPoint1.dx - startPoint.dx).abs();
    final seg2 = (bendPoint2.dy - bendPoint1.dy).abs();
    final seg3 = (endPoint.dx - bendPoint2.dx).abs();
    final totalLen = seg1 + seg2 + seg3;

    final currentDistance = totalLen * progress;
    Offset mechanicPos;
    double headingAngle = 0.0; // in radians

    if (currentDistance <= seg1) {
      final t = currentDistance / seg1;
      mechanicPos = Offset.lerp(startPoint, bendPoint1, t)!;
      headingAngle = 0.0; // moving right
    } else if (currentDistance <= seg1 + seg2) {
      final t = (currentDistance - seg1) / seg2;
      mechanicPos = Offset.lerp(bendPoint1, bendPoint2, t)!;
      headingAngle = math.pi / 2; // moving down
    } else {
      final t = (currentDistance - seg1 - seg2) / seg3;
      mechanicPos = Offset.lerp(bendPoint2, endPoint, t)!;
      headingAngle = 0.0; // moving right
    }

    // 7. Draw Mechanic Motorcycle Marker
    canvas.save();
    canvas.translate(mechanicPos.dx, mechanicPos.dy);

    // Pulse wave around mechanic
    final mechPulse = Paint()
      ..color = const Color(0xFFE65100).withOpacity(0.3)
      ..style = PaintingStyle.fill;
    canvas.drawCircle(Offset.zero, 18, mechPulse);

    // Marker Disc
    final mechDisc = Paint()
      ..color = const Color(0xFFE65100)
      ..style = PaintingStyle.fill;
    canvas.drawCircle(Offset.zero, 12, mechDisc);

    // White center ring
    final mechRing = Paint()
      ..color = Colors.white
      ..strokeWidth = 2.0
      ..style = PaintingStyle.stroke;
    canvas.drawCircle(Offset.zero, 12, mechRing);

    // Little directional indicator triangle
    canvas.rotate(headingAngle);
    final arrowPath = Path();
    arrowPath.moveTo(14, 0);
    arrowPath.lineTo(8, -5);
    arrowPath.lineTo(8, 5);
    arrowPath.close();
    canvas.drawPath(arrowPath, Paint()..color = Colors.white);

    canvas.restore();

    // Mechanic Label floating
    final mechLabelPainter = TextPainter(
      text: const TextSpan(
        text: '🏍️ Mechanic Rajesh',
        style: TextStyle(
          color: Colors.white,
          fontSize: 11,
          fontWeight: FontWeight.bold,
          backgroundColor: Color(0xCC000000),
        ),
      ),
      textDirection: TextDirection.ltr,
    );
    mechLabelPainter.layout();
    mechLabelPainter.paint(
      canvas,
      Offset(mechanicPos.dx - 50, mechanicPos.dy - 32),
    );
  }

  @override
  bool shouldRepaint(covariant RoadMapPainter oldDelegate) {
    return oldDelegate.progress != progress || oldDelegate.status != status;
  }
}
